"""
DownloadBeast_GUI.py

Unified GUI for:
 - downloading videos (yt-dlp)
 - extracting subtitles / forcing transcription (whisper / SpeechRecognition / repo-local)
 - translation / rewrite via repo-local translators or Anthropic/Claude (modules/ai_claude.py)
 - cloud TTS synthesis (Google Cloud or ElevenLabs) with disk caching (modules/tts_cloud.py)
 - quick playback and file open

Notes:
 - This script expects optional helper modules in ./modules:
     - modules/tts_cloud.py     (synthesize_text)
     - modules/ai_claude.py     (rewrite_for_tts)
     - modules/translate_helpers*.py  (local translators)
     - modules/transcribe_translate.py (local transcribe helper)
 - Environment variables:
     - GOOGLE_APPLICATION_CREDENTIALS (Google Cloud service account JSON) for Google TTS
     - ANTHROPIC_API_KEY for Claude (if used)
     - ELEVENLABS_API_KEY for ElevenLabs (if used)
 - Requirements (recommended):
     pip install PySimpleGUI yt-dlp google-cloud-texttospeech requests anthropic sounddevice scipy numpy SpeechRecognition whisper
   (whisper and large deps are optional)
 - Behavior:
     - Download a video (or point to local file).
     - If subtitles available, you can extract them; otherwise use transcription path.
     - Use Claude (optional) to translate/rewrite; then synthesize via selected TTS backend.
     - Synthesized audio is cached (./tts_cache) keyed by hash(text+voice+opts).
 - Backup: this file is intended to replace an existing DownloadBeast_GUI.py; BACKUP before overwriting.
"""

def _import_module_from_path(name: str, path: str):
    try:
        spec = importlib.util.spec_from_file_location(name, path)
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
        return mod
    except Exception:
        return None

# local translators discovery
def discover_local_translators(modules_dir="modules"):
    local_translators = {}
    local_labels = {}
    for path in glob.glob(os.path.join(os.getcwd(), modules_dir, "translate_helpers*.py")):
        fname = os.path.basename(path)
        if fname == "translate_helpers.py":
            code = "local"
            label = "Local helper"
        else:
            if fname.startswith("translate_helpers_"):
                suffix = fname[len("translate_helpers_"):-3]
            else:
                suffix = fname[:-3]
            code = suffix.lower().strip()
            label = suffix.replace("_", " ").title() + " (local)"
        mod = _import_module_from_path(f"local_translate_{code}", path)
        if not mod:
            continue
        translator = None
        if hasattr(mod, "translate_text") and callable(getattr(mod, "translate_text")):
            translator = getattr(mod, "translate_text")
        else:
            for fn in (f"translate_to_{code}", f"translate_{code}", f"to_{code}", "translate"):
                if hasattr(mod, fn) and callable(getattr(mod, fn)):
                    translator = getattr(mod, fn)
                    break
        if translator:
            local_translators[code] = translator
            local_labels[code] = label
    return local_translators, local_labels

# basic language label helper
def prettify(code: str) -> str:
    lookup = {"en":"English","ja":"Japanese","es":"Spanish","fr":"French","de":"German","pt":"Portuguese","zh":"Chinese"}
    if code in lookup: return lookup[code]
    base = code.split("-")[0]
    if base in lookup: return lookup[base]
    return code.replace("_"," ").title()

# helper: run yt-dlp to download video/audio and extract subtitles
def yt_download(url: str, dest_dir: str, extract_subs: bool=False, format_code: str="bestaudio"):
    """
    Returns dict: {"ok":True, "video":video_path, "subs":subtitle_path or None, "err": msg}
    """
    cmd = ["yt-dlp", url, "-f", format_code, "-o", os.path.join(dest_dir, "%(title)s.%(ext)s")]
    if extract_subs:
        # ask yt-dlp for all subtitles (auto/auto-generated)
        cmd += ["--write-auto-sub", "--write-sub", "--sub-lang", "all", "--convert-subs", "srt"]
    try:
        proc = subprocess.run(cmd, capture_output=True, text=True, check=True)
        # try to find downloaded file name from stdout or list files in dest_dir by recent time
        files = sorted(Path(dest_dir).glob("*"), key=lambda p: p.stat().st_mtime, reverse=True)
        video = None
        subs = None
        for f in files:
            if f.suffix.lower() in (".mp4", ".mkv", ".webm", ".m4a", ".mp3", ".wav"):
                video = str(f)
                break
        for f in files:
            if f.suffix.lower() in (".srt", ".vtt"):
                subs = str(f)
                break
        return {"ok":True, "video":video, "subs":subs, "out":proc.stdout}
    except subprocess.CalledProcessError as e:
        return {"ok":False, "err": e.stderr or str(e)}

# transcription utilities
def transcribe_with_local(path: str):
    if LOCAL_TRANSCRIBE:
        try:
            return LOCAL_TRANSCRIBE(path)
        except Exception:
            return None
    return None

def transcribe_with_whisper(path: str, model="small"):
    if not HAVE_WHISPER:
        return None
    try:
        wt = whisper.load_model(model)
        r = wt.transcribe(path)
        return r.get("text", "")
    except Exception:
        return None

def transcribe_with_speechrec(path: str):
    if not HAVE_SPEECHRECOG:
        return None
    try:
        r = sr.Recognizer()
        with sr.AudioFile(path) as source:
            audio = r.record(source)
        return r.recognize_google(audio)
    except Exception:
        return None

def try_transcribe(path: str):
    # try local -> whisper -> speech_rec
    out = transcribe_with_local(path)
    if out:
        return out, "local"
    out = transcribe_with_whisper(path)
    if out:
        return out, "whisper"
    out = transcribe_with_speechrec(path)
    if out:
        return out, "google_sr"
    return None, None

# translation utilities
def try_local_translate(text: str, target_code: str):
    # target_code could be 'ainu' or language code
    if target_code in LOCAL_TRANSLATORS:
        fn = LOCAL_TRANSLATORS[target_code]
        try:
            try:
                return fn(text, target_code)
            except TypeError:
                return fn(text)
        except Exception:
            return None
    if "local" in LOCAL_TRANSLATORS:
        try:
            return LOCAL_TRANSLATORS["local"](text, target_code)
        except Exception:
            return None
    return None

def try_googletrans(text: str, target_code: str):
    if not HAVE_GOOGLETRANS:
        return None
    try:
        tr = GoogleTranslator()
        return tr.translate(text, dest=target_code).text
    except Exception:
        return None

def try_claude_rewrite(text: str, target_lang: str, mode: str = "translate"):
    if AI_CLAUDE is None:
        return None
    try:
        return AI_CLAUDE.rewrite_for_tts(text, target_lang=target_lang, mode=mode)
    except Exception:
        return None

# TTS synth wrapper (calls modules.tts_cloud.synthesize_text)
def synthesize_text_via_cloud(backend: str, text: str, voice: str, fmt: str="mp3", rate: float=1.0, api_key: str=None, cache: bool=True):
    if TTS_CLOUD is None:
        raise RuntimeError("modules/tts_cloud.py not available")
    # write temp text file
    fd, tf = tempfile.mkstemp(prefix="beast_text_", suffix=".txt", text=True)
    os.close(fd)
    try:
        with open(tf, "w", encoding="utf-8") as f:
            f.write(text)
        out = TTS_CLOUD.synthesize_text(backend=backend, text_file=tf, out_path=None, voice=voice, fmt=fmt, rate=rate, api_key=api_key, cache=cache)
        return out
    finally:
        try:
            os.remove(tf)
        except Exception:
            pass

# helper to compute a small cache-key for preview (not used for tts_cloud internal cache)
def _hash_text(text: str, voice: str, fmt: str, backend: str, rate: float):
    key = json.dumps({"text_sha256": hashlib.sha256(text.encode("utf-8")).hexdigest(), "voice":voice or "", "fmt":fmt, "backend":backend, "rate":float(rate)}, sort_keys=True)
    return hashlib.sha256(key.encode("utf-8")).hexdigest()

# background worker to stream subprocess output into GUI safely
def enqueue_subproc(proc, out_q):
    try:
        for line in iter(proc.stdout.readline, b''):
            if not line:
                break
            out_q.put(line.decode(errors='replace'))
    finally:
        try:
            proc.stdout.close()
        except Exception:
            pass
