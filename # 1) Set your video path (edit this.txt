# 1) Set your video path (edit this line)
$file = "C:\Users\kaise\Downloads\ct2_vid_u06_lecture_ppt.mp4"

# 2) Run the pipeline (outputs to ...\videos\processed\myvideo.withsubs.mkv)
powershell -ExecutionPolicy Bypass -Command {
  $PSCommandPath = $null
  $script = @'
Param(
  [Parameter(Mandatory=$true)] [string]$InputFile,
  [string]$OutDir = "C:\Users\kaise\OneDrive\Desktop\videos\processed",
  [string[]]$Langs = @("en","ja"),  # first is treated as "original track label"
  [string]$WhisperModel = "small"   # small|medium|large|... (installed model)
)

# --- Prep paths ---
$InputFile = (Resolve-Path $InputFile).Path
if (-not (Test-Path $OutDir)) { New-Item -ItemType Directory -Path $OutDir | Out-Null }
$nameNoExt = [IO.Path]::GetFileNameWithoutExtension($InputFile)
$ext       = [IO.Path]::GetExtension($InputFile)
$origSrt   = Join-Path $OutDir "$nameNoExt.srt"
$jaSrt     = Join-Path $OutDir "$nameNoExt.ja.srt"
$outMkv    = Join-Path $OutDir "$nameNoExt.withsubs.mkv"

Write-Host "▶ Input : $InputFile"
Write-Host "▶ OutDir: $OutDir"

# --- 1) Transcribe with Whisper CLI (auto language) ---
$whisper = Get-Command whisper -ErrorAction SilentlyContinue
if (-not $whisper) {
  throw "Whisper CLI not found. Activate your venv where whisper is installed, or `pip install -U openai-whisper`."
}
Write-Host "⏳ Transcribing with Whisper ($WhisperModel)..."
& $whisper.Path $InputFile --model $WhisperModel --task transcribe --output_format srt --language auto --output_dir $OutDir 2>$null | Out-Null

# Whisper names the file with original basename; normalize to $origSrt if needed
if (-not (Test-Path $origSrt)) {
  $maybe = Get-ChildItem -LiteralPath $OutDir -Filter "$nameNoExt*.srt" | Select-Object -First 1
  if ($maybe) { Copy-Item $maybe.FullName $origSrt -Force }
}
if (-not (Test-Path $origSrt)) { throw "No SRT produced by Whisper at $origSrt" }
Write-Host "✅ Original SRT: $origSrt"

# --- 2) Translate to Japanese (deep_translator) ---
# (Skip if JA is already same as first track and user didn't request it)
if ($Langs -contains "ja") {
  Write-Host "⏳ Translating to Japanese..."
  $py = Get-Command python -ErrorAction SilentlyContinue
  if (-not $py) { throw "Python not found on PATH. Ensure your Python/venv is active." }

  $tmpPy = [IO.Path]::Combine([IO.Path]::GetTempPath(), "translate_srt_" + [guid]::NewGuid().ToString() + ".py")
  @'
import sys, io, re
from deep_translator import GoogleTranslator

src, dst, tgt = sys.argv[1], sys.argv[2], sys.argv[3]  # e.g., ja
tr = GoogleTranslator(source='auto', target=tgt)

def is_ts(line): return '-->' in line
def is_idx(line): return line.strip().isdigit()

with io.open(src, 'r', encoding='utf-8', errors='ignore') as f:
    lines = f.read().splitlines()

out = []
for line in lines:
    if not line.strip() or is_ts(line) or is_idx(line):
        out.append(line)
    else:
        out.append(tr.translate(line))

with io.open(dst, 'w', encoding='utf-8') as f:
    f.write("\n".join(out))
' | Out-File -FilePath $tmpPy -Encoding UTF8

  & $py.Path $tmpPy $origSrt $jaSrt "ja"
  if (-not (Test-Path $jaSrt)) { throw "Failed to create Japanese SRT at $jaSrt" }
  Write-Host "✅ Japanese SRT: $jaSrt"
}

# --- 3) Mux selectable subtitles into MKV (best container for multi-subs) ---
$ff = Get-Command ffmpeg -ErrorAction SilentlyContinue
if (-not $ff) { throw "ffmpeg not found. Install and add to PATH." }

# Build -i inputs and -map for all subtitle files found
$subs = @($origSrt)
if (Test-Path $jaSrt) { $subs += $jaSrt }

$args = @("-i", $InputFile)
foreach ($s in $subs) { $args += @("-i", $s) }

# maps: 0 = original A/V, 1..N = subs
$args += @("-map","0","-c:v","copy","-c:a","copy","-c:s","srt")

# language tags
for ($i=0; $i -lt $subs.Count; $i++) {
  $lang = ($i -eq 0) ? $Langs[0] : "ja"
  if (-not $lang) { $lang = "und" }
  $args += @("-map", (1 + $i))
  $args += @("-metadata:s:s:$i", "language=$lang")
}

$args += $outMkv

Write-Host "⏳ Muxing to MKV with selectable subs..."
& $ff.Path @args
if (-not (Test-Path $outMkv)) { throw "Failed to create output: $outMkv" }
Write-Host "🎉 Done: $outMkv"
'@

  $tmp = [IO.Path]::GetTempFileName().Replace(".tmp",".ps1")
  $script | Set-Content -Path $tmp -Encoding UTF8
  & powershell -ExecutionPolicy Bypass -File $tmp -InputFile $using:file
}
